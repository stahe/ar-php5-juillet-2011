    3.1. مثال أول

    فيما يلي برنامج يعرض الميزات الأولى لـ PHP.

    3.1.1. البرنامج (exemple_01)

    
<?php

// هذا تعليق
// متغير مستخدم دون أن يتم إعلانه
$nom = "dupont";

// عرض على الشاشة
print "nom=$nom\n";

// مصفوفة تحتوي على عناصر من أنواع مختلفة
$tableau = array("un", "deux", 3, 4);

// عدد عناصره
$n = count($tableau);

// حلقة
for ($i = 0; $i < $n; $i++)
  print "tableau[$i]=$tableau[$i]\n";

// تهيئة متغيرين بمحتوى مصفوفة
list($chaine1, $chaine2) = array("chaine1", "chaine2");

// تسلسل السلسلتين
$chaine3 = $chaine1 . $chaine2;

// عرض النتيجة
print "[$chaine1,$chaine2,$chaine3]\n";

// استخدام دالة
affiche($chaine1);

// يمكن معرفة نوع المتغير
afficheType($n);
afficheType($chaine1);
afficheType($tableau);

// يمكن أن يتغير نوع المتغير أثناء التنفيذ
$n = "a changé";
afficheType($n);

// يمكن للدالة أن تُرجع نتيجة
$res1 = f1(4);
print "res1=$res1\n";

// يمكن للدالة أن ترجع مصفوفة من القيم
list($res1, $res2, $res3) = f2();
print "(res1,res2,res3)=[$res1,$res2,$res3]\n";

// كان من الممكن استرداد هذه القيم في مصفوفة
$t = f2();
for ($i = 0; $i < count($t); $i++)
  print "t[$i]=$t[$i]\n";

// اختبارات
for ($i = 0; $i < count($t); $i++)
// يعرض القنوات فقط
  if (getType($t[$i]) == "string")
    print "t[$i]=$t[$i]\n";

// اختبارات أخرى
for ($i = 0; $i < count($t); $i++){
// يعرض فقط الأعداد الصحيحة الأكبر من 10
  if (getType($t[$i]) == "integer" and $t[$i] > 10)
    print "t[$i]=$t[$i]\n";
}

// حلقة while
$t = array(8, 5, 0, -2, 3, 4);
$i = 0;
$somme = 0;
while ($i < count($t) and $t[$i] > 0) {
  print "t[$i]=$t[$i]\n";
  $somme+=$t[$i];   //$somme=$somme+$t[$i]
  $i++;             //$i=$i+1
}
print "somme=$somme\n";

// نهاية البرنامج
exit;

//----------------------------------
function affiche($chaine) {
  // يعرض $chaine
  print "chaine=$chaine\n";
}

//----------------------------------
function afficheType($variable) {
  // يعرض نوع $variable
  print "type[$variable]=" . getType($variable) . "\n";
}

//----------------------------------
function f1($param) {
  // يضيف 10 إلى $param
  return $param + 10;
}

//----------------------------------
function f2() {
  // تُرجع 3 قيم
  return array("un", 0, 100);
}
?>

    تعليقات

    • السطر 5: في PHP، لم يتم تحديد نوع المتغيرات. هذه المتغيرات لها نوع ديناميكي يمكن أن يتغير بمرور الوقت
    • $nom تمثل المتغير ذي المعرف nom
    • السطر 8: للكتابة على الشاشة، يمكن استخدام الأمر print أو الأمر echo
    • السطر 11: تسمح الكلمة الرئيسية array بتعريف مصفوفة. المتغير $nom[$i] يمثل العنصر $i في المصفوفة $tableau.
    • السطر 14: تُرجع الدالة count($tableau) عدد عناصر المصفوفة $tableau
    • السطر 18: السلاسل محاطة بعلامات اقتباس " أو علامات اقتباس مائلة '. داخل علامات الاقتباس، يتم تقييم المتغيرات $variable ولكن ليس داخل علامات الاقتباس المائلة.
    • السطر 21: تتيح الدالة list تجميع المتغيرات في قائمة وتعيين قيمة لها بعملية تعيين واحدة. هنا $chaine1="chaine1" و $chaine2="chaine2".
    • السطر 24: العلامة . هي علامة ربط السلاسل.
    • السطر 88: الكلمة الرئيسية function تُعرّف دالة. تُرجع الدالة قيمًا أو لا تُرجعها بواسطة الأمر return. يمكن للكود المستدعي تجاهل نتائج الدالة أو استردادها. يمكن تعريف الدالة في أي مكان في الكود.
    • السطر 90: الدالة المُعرَّفة مسبقًا getType($variable) تُرجع سلسلة أحرف تمثل نوع $variable. وقد يتغير هذا النوع بمرور الوقت.
    • السطر 79: الدالة المُعرَّفة مسبقًا exit تُوقف تشغيل البرنامج النصي.

    3.2. نطاق المتغيرات

    3.2.1. البرنامج 1 (exemple_02)

    
<?php

// نطاق المتغيرات

function f1() {
  // يتم استخدام المتغير العام $i
  $i = &$GLOBALS["i"];
  $i++;
  $j = 10;
  print "f1[i,j]=[$i,$j]\n";
}

function f2() {
  // يتم استخدام المتغير العام $i
  $i = &$GLOBALS["i"];
  $i++;
  $j = 20;
  print "f2[i,j]=[$i,$j]\n";
}

function f3() {
  // يتم استخدام متغير محلي $i
  $i = 4;
  $j = 30;
  print "f3[i,j]=[$i,$j]\n";
}

// اختبارات
$i = 0;
$j = 0;  // هاتان المتغيرتان معروفتان لدالة f عبر المصفوفة $GLOBALS
f1();
f2();
f3();
print "test[i,j]=[$i,$j]\n";
?>

    تعليقات

    • السطران 29-30: يحددان متغيرين $i و $j من البرنامج الرئيسي. هذه المتغيرات غير معروفة داخل الدوال. وبالتالي، في السطر 9، فإن المتغير $j الخاص بالدالة f1 هو متغير محلي للدالة f1 ويختلف عن المتغير $j في البرنامج الرئيسي. يمكن لدالة الوصول إلى متغير $variable في البرنامج الرئيسي عبر مصفوفة من المتغيرات العالمية تسمى $GLOBALS. في السطر 7، تشير الصيغة $GLOBALS["i"] إلى المتغير العام $i في البرنامج الرئيسي. إذا كتبنا 
    $i=$GLOBALS["i"]

    يتم تعيين قيمة المتغير العام $i إلى المتغير المحلي $i. وهنا لدينا متغيران مختلفان، ولن يؤدي تعديل المتغير المحلي $i إلى تغيير المتغير العام $i. الترميز

    $i=&Globals["i"]

    يجعل المتغير المحلي $i له نفس عنوان الذاكرة للمتغير العام $i. وبالتالي، فإن التعامل مع المتغير المحلي $i يعادل التعامل مع المتغير العام $i.

$tab1[] = $i++;

    تعليقات

    يُظهر البرنامج أعلاه عمليات معالجة مصفوفة من القيم. توجد طريقتان لكتابة المصفوفات في PHP:

    1
2
    $tableau=array("un",2,"trois")
$contraires=array("petit"=>"grand", "beau"=>"laid", "cher"=>"bon marché")

    يُطلق على المصفوفة 1 اسم «مصفوفة»، أما المصفوفة 2 فتُسمى «قاموس» حيث تُدوّن العناصر على النحو التالي: المفتاح => القيمة. تشير الصيغة $contraires["beau"] إلى القيمة المرتبطة بالمفتاح "جميل". وهي هنا السلسلة "قبيح". المصفوفة 1 ليست سوى شكل آخر من أشكال القاموس ويمكن تدوينها على النحو التالي:

    $tableau=array(0=>"un",1=>2,2=>"trois")

    وبذلك نحصل على $tableau[2]="trois". في النهاية، لا يوجد سوى القواميس. في حالة مصفوفة تقليدية مكونة من n عنصرًا، تكون المفاتيح هي الأعداد الصحيحة من النطاق [0,n-1].

    • السطر 15: تسمح الدالة each($tableau) بالتنقل عبر القاموس. وعند كل استدعاء، تُرجع زوجًا (مفتاح، قيمة) منه.
    • السطر 14: تضع الدالة reset($dictionnaire) الدالة each على الزوج الأول (مفتاح، قيمة) من القاموس.
    • السطر 15: تتوقف الحلقة while عندما تُرجع الدالة each زوجًا فارغًا عند نهاية القاموس.
    • السطر 19: تضيف صيغة $tableau[]=قيمة العنصر valeur كآخر عنصر في $tableau.
    • السطر 25: يتم تصفح المصفوفة باستخدام foreach. يسمح هذا العنصر النحوي بتصفح قاموس، وبالتالي مصفوفة، وفقًا لنظامين نحويين:
    foreach($dictionnaire as $clé=>$valeur)
foreach($tableau as $valeur)

    تُرجع الصيغة الأولى زوجًا (clé,valeur) في كل تكرار، بينما تُرجع الصيغة الثانية العنصر valeur من القاموس فقط.

    • السطر 31: الدالة array_pop($tableau) تحذف العنصر الأخير من $tableau
    • السطر 39: تقوم الدالة array_shift($tableau) بحذف العنصر الأول من $tableau
    • السطر 47: تضيف الدالة array_push($tableau,قيمة) valeur كعنصر أخير في $tableau
    • السطر 39: تضيف الدالة array_unshift($tableau، القيمة) valeur كأول عنصر في $tableau

    3.3.2. القاموس (exemple_05)

    
<?php

// قواميس
$conjoints = array("Pierre" => "Gisèle", "Paul" => "Virginie", "Jacques" => "Lucette", "Jean" => "");

// مسار - 1
print "Nombre d'éléments du dictionnaire : " . count($conjoints) . "\n";
reset($conjoints);
while (list($clé, $valeur) = each($conjoints))
  print "conjoints[$clé]=$valeur\n";

// فرز القاموس حسب المفتاح
ksort($conjoints);

// مسار - 2
reset($conjoints);
while (list($clé, $valeur) = each($conjoints))
  print "conjoints[$clé]=$valeur\n";

// قائمة مفاتيح القاموس
$clés = array_keys($conjoints);
for ($i = 0; $i < count($clés); $i++)
  print "clés[$i]=$clés[$i]\n";

// قائمة قيم القاموس
$valeurs = array_values($conjoints);
for ($i = 0; $i < count($valeurs); $i++)
  print "valeurs[$i]=$valeurs[$i]\n";

// البحث عن مفتاح
existe($conjoints, "Jacques");
existe($conjoints, "Lucette");
existe($conjoints, "Jean");

// حذف مفتاح وقيمة
unset($conjoints["Jean"]);
print "Nombre d'éléments du dictionnaire : " . count($conjoints) . "\n";
foreach ($conjoints as $clé => $valeur) {
  print "conjoints[$clé]=$valeur\n";
}

// النهاية
exit;

function existe($conjoints, $mari) {
  // التحقق من وجود المفتاح $mari في القاموس $conjoints
  if (isset($conjoints[$mari]))
    print "La clé [$mari] existe associée à la valeur [$conjoints[$mari]]\n";
  else
    print "La clé [$mari] n'existe pas\n";
}

?>

    3.3.3. المصفوفات متعددة الأبعاد (exemple_06)

    
<?php

// الجداول التقليدية متعددة الأبعاد
// التهيئة
$multi = array(array(0, 1, 2), array(10, 11, 12, 13), array(20, 21, 22, 23, 24));
// التصفح
for ($i1 = 0; $i1 < count($multi); $i1++)
  for ($i2 = 0; $i2 < count($multi[$i1]); $i2++)
    print "multi[$i1][$i2]=" . $multi[$i1][$i2] . "\n";
// قواميس متعددة الأبعاد

$correspond = preg_match($modèle, $chaîne, $champs);

    تُرجع الدالة preg_match قيمة منطقية إذا كان من الممكن العثور على modèle في chaîne. إذا كان الأمر كذلك، فإن $champs[0] تمثل السلسلة الفرعية المطابقة للنموذج. من ناحية أخرى، إذا كان النموذج يحتوي على نماذج فرعية بين قوسين، فإن $champs[1] هو الجزء من $chaîne المطابق للنموذج الفرعي الأول، و $champs[2] هو الجزء من $chaîne الذي يتوافق مع النموذج الفرعي الثاني، وهكذا دواليك...

    لنأخذ المثال الأول. يتم تعريف النموذج في السطر 8: وهو يشير إلى سلسلة من رقم واحد أو أكثر (+) من الأرقام (\d) موضوعة في أي مكان في السلسلة. علاوة على ذلك، يحدد النموذج نموذجًا فرعيًا محاطًا بأقواس.

    • السطر 10: يتم مقارنة النموذج /(\d+)/ (سلسلة من رقم واحد أو أكثر في أي مكان في السلسلة) بالسلسلة "xyz1234abcd". نرى أن السلسلة الفرعية 1234 تتطابق مع النموذج. وبالتالي، سيكون $champs[0] مساويًا لـ "1234". من ناحية أخرى، يحتوي النمط على أنماط فرعية بين الأقواس. فسنحصل على $champs[1] = "1234".
    • السطر 11: يتم مقارنة النمط /(\d+)/ بالسلسلة "12 34". نلاحظ أن السلسلتين الفرعيتين 12 و34 تتطابقان مع النمط. تتوقف المقارنة عند أول سلسلة فرعية تتطابق مع النمط. وبالتالي، سيكون لدينا $champs[0]=12 و $champs[1]=12.
    • السطر 12: يتم مقارنة النمط /(\d+)/ بالسلسلة "abcd". لم يتم العثور على أي تطابق.

    دعونا نوضح الأنماط المستخدمة في بقية الكود:

    
$modèle = "/^(.*?)(\d+)(.*?)$/";

    بداية السلسلة (^)، ثم صفر أو أكثر (*) من الأحرف العشوائية (.)، ثم رقم واحد أو أكثر (+)، ثم مرة أخرى صفر أو أكثر (*) من الأحرف العشوائية (.). يشير النمط (.*) إلى صفر أو أكثر من الأحرف العشوائية. سيتطابق هذا النمط مع أي سلسلة. وبالتالي، لن يتم العثور أبدًا على النمط /^(.*)(\d+)(.*)$/ لأن النمط الفرعي الأول (.*) سيستوعب السلسلة بأكملها. أما النمط (.*?)(\d+) فيشير إلى صفر أو أكثر من الأحرف العشوائية حتى النمط الفرعي التالي (؟)، وهو هنا \d+. وبالتالي، لم تعد الأرقام تُستوعب الآن بواسطة النمط (.*). وبالتالي، فإن النمط أعلاه يتطابق مع [début de chaîne (^), une suite de caractères quelconques (.*?), une suite d'un ou plusieurs chiffres (\d+), une suite de caractères quelconques (.*?), la fin de la chaîne ($)].

    • &$a: تشير علامة & إلى أن المعلمة الشكلية $a ستأخذ كقيمة عنوان المعلمة الفعلية $i. بعبارة أخرى، $a و $i هما مرجعان لنفس موقع الذاكرة. إن التعامل مع المعلمة الشكلية $a يعادل التعامل مع المعلمة الفعلية $i. وهذا ما يظهره تنفيذ الكود. يناسب هذا النمط من التمرير معلمات الإخراج والبيانات الضخمة مثل المصفوفات والقواميس. ويُطلق على هذا النمط من التمرير اسم «التمرير بالمرجع».
    • $b: ستتخذ المعلمة الشكلية $b القيمة نفسها التي تتخذها المعلمة الفعلية $j. وهذا يمثل تمريرًا بالقيمة. المعلمات الشكلية والفعلية هما متغيران مختلفان. ولا يؤثر التعامل مع المعلمة الشكلية $b بأي شكل من الأشكال على المعلمة الفعلية $j. وهذا ما يوضحه تنفيذ الكود. ويناسب هذا النمط من التمرير معلمات الإدخال.

function afficheInfos($dico, $clé) {
  // عرض القيمة المرتبطة بالمفتاح في القاموس $dico إذا كانت موجودة
  if (isset($dico[$clé])) {
    // القيمة موجودة - هل هي مصفوفة؟
    $valeur = $dico[$clé];
    if (is_array($valeur)) {
      print "[$clé," . join(":", $valeur) . "]\n";
    } else {
      // $valeur ليس مصفوفة
      print "[$clé,$valeur]\n";
    }
  } else {
    // $clé ليس مفتاحًا في القاموس $dico
    print "la clé [$clé] n'existe pas\n";
  }
}

// --------------------------------------------------------------------------
function cutNewLinechar($ligne) {
  // يتم حذف علامة نهاية السطر من $ligne إن وجدت
  $L = strlen($ligne);  // طول السطر
  while (substr($ligne, $L - 1, 1) == "\n" or substr($ligne, $L - 1, 1) == "\r") {
    $ligne = substr($ligne, 0, $L - 1);
    $L--;
  }
  // النهاية
  return($ligne);
}

    • السطر 11: fopen(nom_fichier,"w") يفتح الملف nom_fichier في وضع الكتابة (w=write). إذا لم يكن الملف موجودًا، يتم إنشاؤه. وإذا كان موجودًا، يتم مسحه. وإذا فشل الإنشاء، فإن fopen تُرجع القيمة false. في العبارة if (!$fic = fopen($INFOS, "w")) {…}، توجد عمليتان متتاليتان: 1) $fic=fopen(..) 2) if( ! $fic) {...}
    • السطر 18: fputs($fic,$chaîne) يكتب chaîne في الملف $fic.
    • السطر 20: fclose($fic) يغلق الملف $fic.
    • السطر 25: fopen(nom_fichier,"r") يفتح الملف nom_fichier للقراءة (r=read). إذا فشل فتح الملف (الملف غير موجود)، فإن fopen تُرجع القيمة false.
    • السطر 29: fgets($fic,1000) يقرأ السطر التالي من الملف بحد أقصى 1000 حرف. في العملية while ($ligne = fgets($fic, 1000)) {…}، هناك عمليتان متتاليتان 1) $ligne=fgets(...) 2) while ( ! $ligne). بعد قراءة الحرف الأخير من الملف، تُفرغ الدالة fgets السلسلة وتتوقف الحلقة while.
    • السطر 31: تقرأ الدالة fgets سطرًا من النص، بما في ذلك حرف نهاية السطر. تقوم الدالة cutNewLineChar في الأسطر 70-79 بإزالة حرف نهاية السطر إن وجد.
    • السطر 72: تعرض الدالة strlen($chaîne) عدد أحرف $chaîne.
    • السطر 73: الدالة substr($ligne, $position, $taille) تُرجع $taille حرفًا من $ligne، بدءًا من الحرف رقم $position، حيث يُعتبر الحرف الأول هو رقم 0. على أجهزة ويندوز، علامة نهاية السطر هي "\r\n". على أجهزة يونكس، هي السلسلة "\n".
    • السطر 35: الدالة array_shift($tableau) تحذف العنصر الأول من $tableau وتعيّنه كنتيجة.
    • السطر 37: يتم استخدام الدالة array_shift($tableau) ولكن يتم تجاهل نتيجتها
    • السطر 57: تُرجع الدالة is_array($variable) القيمة true إذا كانت $variable مصفوفة، وإلا تُرجع false.
    • السطر 58: الدالة join تقوم بنفس الشيء الذي تقوم به الدالة implode التي سبق أن تناولناها
