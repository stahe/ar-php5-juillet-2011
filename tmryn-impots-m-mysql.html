    8. تمرين IMPOTS مع MySQL

function copyToMysql($limites, $coeffR, $coeffN, $HOTE, $USER, $PWD, $BASE, $TABLE) {
  // نسخ الجداول الرقمية الثلاثة $limites، $coeffR، $coeffN
  // إلى الجدول $TABLE في قاعدة بيانات mysql $BASE
  // قاعدة بيانات mysql موجودة على الجهاز $HOTE
  // يتم تعريف المستخدم بواسطة $USER و $PWD
  // تسجيل الدخول
  list($erreur, $connexion) = connecte("mysql:host=$HOTE;dbname=$BASE", $USER, $PWD);
  if ($erreur)
    return "Erreur lors de la connexion à MySql sous l'identité ($HOTE,$USER,$PWD) : $erreur\n";
  // حذف الجدول
  $requête = "drop table $TABLE";
  exécuteRequête($connexion, $requête);
  // إنشاء الجدول
  $requête = "create table $TABLE (limites decimal(10,2), coeffR decimal(6,2), coeffN decimal(10,2))";
  list($erreur, $res) = exécuteRequête($connexion, $requête);
  if ($erreur)
    return "$requête : erreur ($erreur)";
  // التعبئة
  for ($i = 0; $i < count($limites); $i++) {
    // استعلام الإدراج
    $requête = "insert into $TABLE (limites,coeffR,coeffN) values ($limites[$i],$coeffR[$i],$coeffN[$i])";
    // تنفيذ الاستعلام
    list($erreur, $res) = exécuteRequête($connexion, $requête);
    // العودة في حالة وجود خطأ
    if ($erreur)
      return "$requête : erreur ($erreur)";
  }//for
  // انتهى الأمر - يتم قطع الاتصال
  déconnecte($connexion);
  // العودة دون أخطاء
  return "";
}

    8.2. برنامج حساب الضريبة ( impots_04)

    تستخدم هذه النسخة فئة Impôts التي تبحث في قاعدة البيانات عن القيم اللازمة لحساب الضريبة. نقدم هنا مفهومًا جديدًا، وهو مفهوم الاستعلام المُعد مسبقًا. يتم تنفيذ الأمر SQL بواسطة SGBD على مرحلتين:

    1. يتم إعداد الاستعلام: يقوم الأمر SGBD بإعداد خطة تنفيذ مُحسَّنة للاستعلام. والهدف من ذلك هو تنفيذه بأكبر قدر ممكن من الكفاءة.
    1. يتم تنفيذ الطلب.

    إذا تم تنفيذ نفس الاستعلام N مرات، يتم تنفيذ المرحلتين السابقتين N مرات. ومع ذلك، من الممكن إعداد الاستعلام مرة واحدة وتنفيذه N مرات. ولتحقيق ذلك، يجب استخدام الاستعلامات المُعدة مسبقًا. إذا كان $requête هو الأمر SQL المطلوب تنفيذه، و$connexion هو الكائن PDO الذي يمثل الاتصال:

    • $statement=$connexion->prepare($requête) تقوم بإعداد استعلام وتجعل الاستعلام "مُعدًا"
    • $statement->execute() تنفذ الاستعلام المُعد.

    إذا كان الاستعلام المُعدّ هو الأمر select، فإن

    • $statement->fetchAll() تُرجع جميع صفوف الجدول الناتج عن select في شكل مصفوفة T حيث T[i,j] هي قيمة العمود j في الصف i من الجدول
    • $statement->fetch() تُرجع السطر الحالي من الجدول في شكل مصفوفة T، حيث T[j] هي قيمة العمود j في السطر

    توفر الاستعلامات المعدة مسبقًا مزايا أخرى بخلاف الكفاءة العالية. فهي توفر على وجه الخصوص مزيدًا من الأمان. لذا ينبغي استخدامها بشكل منهجي.

while ($ligne = fgets($data, 100)) {

  $ligne = $u->cutNewLineChar($ligne);
